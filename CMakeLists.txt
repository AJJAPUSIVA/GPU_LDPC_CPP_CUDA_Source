cmake_minimum_required(VERSION 3.24)
project(nr_gpu_ldpc LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NRLDPC_ENABLE_CUDA "Build NVIDIA-only CUDA codec and tests" OFF)
option(NRLDPC_ENABLE_CPU "Build independent CPU decoder research backend" ON)
enable_testing()
if(NRLDPC_ENABLE_CPU)
  add_library(nrldpc_cpu STATIC src/cpu_decoder.cpp)
  target_include_directories(nrldpc_cpu PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  add_executable(ldpc_cpu_test tests/test_cpu.cpp)
  target_link_libraries(ldpc_cpu_test PRIVATE nrldpc_cpu)
  add_test(NAME cpu_toy_syndrome COMMAND ldpc_cpu_test ${CMAKE_CURRENT_SOURCE_DIR}/matrices/toy_qc_z4.txt)
  add_executable(ldpc_modes_test tests/test_decoder_modes.cpp)
  target_link_libraries(ldpc_modes_test PRIVATE nrldpc_cpu)
  add_test(NAME cpu_modes COMMAND ldpc_modes_test ${CMAKE_CURRENT_SOURCE_DIR}/matrices/toy_qc_z4.txt)
  add_executable(ldpc_awgn_bench benchmarks/benchmark_awgn.cpp)
  target_link_libraries(ldpc_awgn_bench PRIVATE nrldpc_cpu)
  add_test(NAME cpu_awgn_fixed COMMAND ldpc_awgn_bench ${CMAKE_CURRENT_SOURCE_DIR}/matrices/toy_qc_z4.txt 19 5 2.0 fixed scalar 123)
  add_executable(ldpc_replay_cpu benchmarks/replay_cpu.cpp)
  target_link_libraries(ldpc_replay_cpu PRIVATE nrldpc_cpu)
  add_executable(ldpc_cpu_bench benchmarks/benchmark_cpu.cpp)
  target_link_libraries(ldpc_cpu_bench PRIVATE nrldpc_cpu)
endif()
if(NRLDPC_ENABLE_CUDA)
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 20)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 89)
  endif()
  find_package(CUDAToolkit REQUIRED)
  add_library(nrldpc STATIC src/ldpc.cpp src/ldpc.cu)
  target_include_directories(nrldpc PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_link_libraries(nrldpc PUBLIC CUDA::cudart)
  set_target_properties(nrldpc PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
  target_compile_options(nrldpc PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
  add_executable(ldpc_test tests/test_codec.cu)
  target_link_libraries(ldpc_test PRIVATE nrldpc)
  add_executable(ldpc_awgn_cuda_bench benchmarks/benchmark_awgn_cuda.cu)
  target_link_libraries(ldpc_awgn_cuda_bench PRIVATE nrldpc)
  add_executable(ldpc_replay_cuda benchmarks/replay_cuda.cu)
  target_link_libraries(ldpc_replay_cuda PRIVATE nrldpc)
  add_executable(ldpc_bench benchmarks/benchmark.cu)
  target_link_libraries(ldpc_bench PRIVATE nrldpc)
  add_test(NAME toy_qc_gpu_roundtrip COMMAND ldpc_test ${CMAKE_CURRENT_SOURCE_DIR}/matrices/toy_qc_z4.txt)
endif()

# Experimental CUDA workspace A/B benchmark
if(CMAKE_CUDA_COMPILER)
    add_executable(ldpc_workspace_ab benchmarks/workspace_ab.cu)
    target_link_libraries(ldpc_workspace_ab PRIVATE nrldpc)
    set_target_properties(ldpc_workspace_ab PROPERTIES
        CUDA_SEPARABLE_COMPILATION ON)
endif()

# Saved-AWGN workspace A/B benchmark
if(CMAKE_CUDA_COMPILER)
  add_executable(ldpc_workspace_awgn_ab
    benchmarks/workspace_awgn_ab.cu)
  target_link_libraries(ldpc_workspace_awgn_ab PRIVATE nrldpc)
  set_target_properties(ldpc_workspace_awgn_ab PROPERTIES
    CUDA_SEPARABLE_COMPILATION ON)
endif()
